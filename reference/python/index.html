    def validate_combination(selection, config):
    contract = config["training"]["backend_contract"]
    if contract not in selection.model.training_backend_contracts:
        raise ValueError(
            f"model {selection.model.id} does not support training backend {contract}"
        )
    if contract not in selection.algorithm.training_backend_contracts:
        raise ValueError(
            f"algorithm {selection.algorithm.id} does not support training backend {contract}"
        )
    if config["execution"]["backend"] not in selection.task.execution_backends:
        raise ValueError(
            f"task provider {selection.task.id} does not support execution backend "
            f"{config['execution']['backend']}"
        )
    if config["harbor"]["version"] not in selection.harness.harbor_versions:
        raise ValueError(
            f"harness {selection.harness.id} does not support Harbor "
            f"{config['harbor']['version']}"
        )
    if selection.model.tool_protocol != selection.harness.tool_protocol:
        raise ValueError("model and harness tool protocols do not match")
    if selection.harness.request_protocol not in selection.model.gateway_protocols:
        raise ValueError(
            f"model gateway does not support harness protocol {selection.harness.request_protocol}"
        )
    if config["sampling"]["group_size"] < selection.algorithm.minimum_group_size:
        raise ValueError(
            f"algorithm {selection.algorithm.id} requires group size >= "
            f"{selection.algorithm.minimum_group_size}"
        )
    if (
        selection.algorithm.requires_audited_raw_logprobs
        and not config["gateway"]["audited_raw_logprobs"]
    ):
        raise ValueError(
            f"algorithm {selection.algorithm.id} requires audited raw logprobs"
        )
    return selection

    provider_id = "native_harbor"

    def discover(self, tasks_dir):
        return sorted(path.parent for path in Path(tasks_dir).rglob("task.toml"))

    def inspect(self, path, *, expected_digest=None):
        return inspect_native(path, expected_digest=expected_digest)

    def catalog(self, config):
        path = Path(config["tasks"]["catalog"])
        entries = read_json(path)
        if not isinstance(entries, list) or not entries:
            raise ValueError("catalog must be a nonempty JSON list")
        seen = set()
        for entry in entries:
            if (
                not isinstance(entry, dict)
                or set(entry)
                != {"id", "revision", "path", "task_digest", "reward_profile"}
                or not isinstance(entry["id"], str)
                or not entry["id"].strip()
                or not isinstance(entry["revision"], str)
                or not entry["revision"].strip()
                or entry["id"] in seen
            ):
                raise ValueError("invalid or duplicate catalog identity")
            if (
                not isinstance(entry["path"], str)
                or not entry["path"].strip()
                or not isinstance(entry["task_digest"], str)
                or not re.fullmatch(r"[0-9a-f]{64}", entry["task_digest"])
                or not isinstance(entry["reward_profile"], dict)
            ):
                raise ValueError(f'invalid native task lock: {entry["id"]}')
            seen.add(entry["id"])
            entry["path"] = str((path.parent / entry["path"]).resolve())
            report = self.inspect(entry["path"], expected_digest=entry["task_digest"])
            if report["status"] != "NEEDS_PROBE":
                raise ValueError(
                    f'task preflight failed: {entry["id"]}: {report["reasons"]}'
                )
            RewardProfile(**entry["reward_profile"])
        selected = set(getattr(config, "task_ids", ()))
        if selected:
            missing = selected - {entry["id"] for entry in entries}
            if missing:
                raise ValueError(f"task selector not found in catalog: {min(missing)}")
            entries = [entry for entry in entries if entry["id"] in selected]
        return entries

    def prompt_rows(self, entries):
        rows = []
        for entry in entries:
            instruction = (Path(entry["path"]) / "instruction.md").read_text()
            rows.append(

    @dataclass(frozen=True)
class RewardProfile:
    key: str = "reward"
    scale: float = 1.0
    offset: float = 0.0
    raw_range: tuple[float, float] | None = (0.0, 1.0)

    def __post_init__(self):
        if not isinstance(self.key, str) or not self.key:
            raise ValueError("reward key is required")
        if not finite(self.scale) or not finite(self.offset):
            raise ValueError("reward transform must be finite")
        if self.raw_range is not None:
            if (len(self.raw_range) != 2 or not all(finite(x) for x in self.raw_range)
                    or self.raw_range[0] > self.raw_range[1]):
                raise ValueError("invalid raw reward range")
            object.__setattr__(self, "raw_range", tuple(self.raw_range))

    @property
    def digest(self):
        return digest(asdict(self))

    def resolve(self, rewards):
        if not isinstance(rewards, dict) or not rewards or any(not finite(v) for v in rewards.values()):
            raise ValueError("all raw rewards must be finite numbers, not bools")
        if self.key not in rewards:
            raise ValueError(f"missing selected reward key: {self.key}")
        raw = rewards[self.key]
        if self.raw_range is not None and not self.raw_range[0] <= raw <= self.raw_range[1]:
            raise ValueError("reward outside declared range")
        training = self.scale * raw + self.offset
        if not finite(training):
            raise ValueError("nonfinite transformed reward")
        return {"raw_rewards": rewards, "selected_key": self.key, "raw_reward": raw,
                "training_reward": training, "reward_profile": json.loads(encode(asdict(self))),
                "reward_profile_digest": self.digest}

    def publish(path, value):
    """Publish once, idempotently; concurrent conflicting writers never overwrite."""
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    payload = encode(value) + b"\n"
    fd, temp = tempfile.mkstemp(prefix=".pending-", dir=path.parent)
    try:
        with os.fdopen(fd, "wb") as stream:
            stream.write(payload)
            stream.flush()
            os.fsync(stream.fileno())
        try:
            os.link(temp, path)
        except FileExistsError:
            if path.read_bytes() != payload:
                raise ValueError(f"conflicting immutable record: {path.name}")
        if os.name == "posix":
            directory = os.open(path.parent, os.O_RDONLY)
            try:
                os.fsync(directory)
            finally:
                os.close(directory)
    finally:
        os.unlink(temp)
    return path

    def validate_v2(ir):
    """Recompute eligibility; never trust a caller-supplied readiness flag."""
    errors = []
    try:
        identity = Identity(**ir["identity"])
        trial = ir["harbor_trial_id"]
        if ir.get("schema_version") != "2" or not isinstance(trial, str) or not trial:
            errors.append("schema_or_trial")
        receipt, seal, turns = ir["evaluation"], ir["trace_seal"], ir["turns"]
        if receipt.get("identity") != identity.to_dict() or receipt.get("harbor_trial_id") != trial:
            errors.append("evaluation_identity")
        if receipt.get("evaluation_status") != "valid":
            errors.append("evaluation_status")
        profile = RewardProfile(**receipt["reward_profile"])
        resolved = profile.resolve(receipt["raw_rewards"])
        if profile.digest != identity.reward_digest or any(receipt.get(k) != v for k, v in resolved.items()):
            errors.append("reward_recompute")
        if not receipt.get("source_result_digest") or not receipt.get("source_artifact_digest"):
            errors.append("evaluation_evidence")
        if seal.get("identity") != identity.to_dict() or seal.get("harbor_trial_id") != trial:
            errors.append("seal_identity")
        if seal.get("pending_requests") != 0 or seal.get("status") != "sealed" or seal.get("errors"):
            errors.append("trace_not_sealed")
        if not isinstance(turns, list) or not turns:
            errors.append("empty_turns")
            return errors
        request_ids = [t["request_id"] for t in turns]

    def group_statistics(rewards, *, epsilon=1e-6):
    """Population mean/std and GRPO advantages for one complete group."""
    values = list(rewards)
    if len(values) < 2:
        raise ValueError("a complete group of at least two trajectories is required")
    if any(not finite(value) for value in values):
        raise ValueError("nonfinite group statistics")
    if not finite(epsilon) or epsilon <= 0:
        raise ValueError("epsilon must be positive and finite")
    mean = math.fsum(values) / len(values)
    std = math.sqrt(math.fsum((value - mean) ** 2 for value in values) / len(values))
    if not finite(mean) or not finite(std):
        raise ValueError("nonfinite group statistics")
    advantages = [0.0 if std == 0 else (value - mean) / (std + epsilon) for value in values]
    return mean, std, advantages

    """Validate complete Native IR groups and export immutable training units."""
    groups = [list(group) for group in ir_groups]
    if not groups:
        raise ValueError("native training batch requires at least one group")
    ids = _rollout_ids(groups, rollout_ids, rollout_id_base)
    exported_groups = []
    policy_versions = set()
    for trajectories, rollout_id in zip(groups, ids):
        if group_size is None:
            expected_size = len(trajectories)
        else:
            expected_size = group_size
            if type(expected_size) is not int:
                raise ValueError("group_size must be an integer")
        if expected_size < 2 or len(trajectories) != expected_size:
            raise ValueError("a complete group of at least two trajectories is required")

            require_ready(ir, allow_synthetic=allow_synthetic)

            raise ValueError("mixed group, policy, task or profiles")

                raise ValueError(f"duplicate {key}")

        mean, std, advantages = group_statistics(rewards, epsilon=epsilon)
        units = []

            turns = []
            token_count = 0
            for turn_index, turn in enumerate(ir["turns"]):
                prompt = turn["input_ids"]
                response = turn["output_ids"]
                turns.append(TokenSpan(
                    turn_index=turn_index,
                    tokens=prompt + response,
                    prompt_length=len(prompt),
                    response_length=len(response),
                    loss_mask=[False] * len(prompt) + [True] * len(response),
                    old_logprobs=turn["logprobs"],
                    policy_version=identity.policy_version,
                ))
                token_count += len(response)
            units.append(TrajectoryTrainingUnit(
                trajectory_id=identity.trajectory_id,
                rollout_id=rollout_id,
                group_key=identity.group_key,
                reward=float(reward),
                advantage=float(advantage),
                token_count=token_count,
                turns=tuple(turns),
            ))
            policy_versions.add(identity.policy_version)
        exported_groups.append(TrainingGroup(
            group_key=identities[0].group_key,
            group_size=expected_size,
            reward_mean=mean,
            reward_std=std,
            trajectories=tuple(units),

        schema_version=SCHEMA_VERSION,
        groups=tuple(exported_groups),
        reduction=REDUCTION,
        policy_versions=frozenset(policy_versions),
    )
    validate_batch(batch, epsilon=epsilon)

    def run_embedded(coroutine):
    """Run successive synchronous rollout calls on one Python 3.10-compatible loop."""
    global _EMBEDDED_LOOP, _EMBEDDED_THREAD
    try:
        asyncio.get_running_loop()

        pass
    else:
        coroutine.close()

    with _EMBEDDED_LOCK:
        if _EMBEDDED_LOOP is None:
            _EMBEDDED_LOOP = asyncio.new_event_loop()

    def close_runtime():
    """Drain embedded tasks, close the gateway, then release the owning loop."""
    global _RUNTIME, _EMBEDDED_LOOP, _EMBEDDED_THREAD
    with _EMBEDDED_LOCK:
        loop = _EMBEDDED_LOOP
        if loop is not None and loop.is_running():
            raise RuntimeLifecycleError("close embedded rollout between calls")
        if loop is not None:
            try:
                asyncio.get_running_loop()
            except RuntimeError:
                pass
            else:
                raise RuntimeLifecycleError("close embedded rollout outside an active event loop")

        async def cancel_pending():
            pending = asyncio.all_tasks() - {asyncio.current_task()}
            for task in pending:
                task.cancel()
            await asyncio.gather(*pending, return_exceptions=True)

        try:
            if loop is not None:
                loop.run_until_complete(cancel_pending())
            with _RUNTIME_LOCK:
                cached, _RUNTIME = _RUNTIME, None
                try:
                    if cached is not None:
                        cached.close()
                finally:
                    if loop is not None:
                        loop.run_until_complete(loop.shutdown_asyncgens())
                        loop.run_until_complete(loop.shutdown_default_executor())
        finally:
            if loop is not None:
                loop.close()
            _EMBEDDED_LOOP = None
            _EMBEDDED_THREAD = None

    class LocalTrainingTarget:
    """Legacy target: the Slime job runs as a child of this process."""

    def __init__(self, command, *, cwd, env):
        self.command = [str(part) for part in command]
        self.cwd = str(cwd)
        self.env = env
        self.child = None

    def launch(self):
        self.child = subprocess.Popen(self.command, cwd=self.cwd, env=self.env,
                                      start_new_session=True)
        return self.child.pid

    def wait(self):
        if self.child is None:
            raise TrainingTargetError("local target was never launched")

        def forward_signal(signum, frame):
            if self.child.poll() is None:
                try:
                    os.killpg(os.getpgid(self.child.pid), signum)
                except ProcessLookupError:
                    pass

        previous = {sig: signal.signal(sig, forward_signal)
                    for sig in (signal.SIGINT, signal.SIGTERM)}
        try:
            return self.child.wait()
        finally:
            for sig, handler in previous.items():
                signal.signal(sig, handler)

    def query(self):
        if self.child is None:
            return {"state": "not-launched"}
        return {"state": "running" if self.child.poll() is None else "exited",
                "exit_code": self.child.returncode}

    def cancel(self):
        if self.child is None or self.child.poll() is not None:
            return False
        try:
            os.killpg(os.getpgid(self.child.pid), signal.SIGINT)
        except (ProcessLookupError, PermissionError):
            self.child.terminate()
        return True

    def launch(self):
        if self._launched:
            raise TrainingTargetError("ssh target was already launched")
        # The background job records its own session-leader pid: setsid execs
        # the launcher, so the recorded pid heads the cancellable process group.
        bootstrap = (f"cd {shlex.quote(str(self.run_dir))} && "
                     f"(setsid nohup bash {LAUNCHER_NAME} > {STDOUT_NAME} 2>&1"
                     f" < /dev/null & echo $! > {PID_NAME})")
        result = remote.run_ssh(self.spec, bootstrap, timeout=self.command_timeout,
                                connect_timeout=self.ssh_timeout)
        if result.returncode != 0:
            raise TrainingTargetError(
                f"remote launch failed on {destination(self.spec)}: "
                f"{result.stderr.strip()[-500:]}")
        self._launched = True
        return self.run_dir / PID_NAME

    def _pid(self):
        try:
            return int((self.run_dir / PID_NAME).read_text().strip())
        except (OSError, ValueError):
            return None

    def exit_code(self):
        try:
            return int((self.run_dir / EXIT_CODE_NAME).read_text().strip())
        except (OSError, ValueError):
            return None

    def alive(self):
        pid = self._pid()
        if pid is None:
            return False
        result = remote.run_ssh(self.spec, f"kill -0 {pid} 2>/dev/null && echo ok",
                                timeout=self.command_timeout, connect_timeout=self.ssh_timeout)
        return result.returncode == 0 and result.stdout.strip() == "ok"

    def wait(self, *, timeout=None):
        """Poll the shared exit-code record; never resubmit or restart."""
        deadline = None if timeout is None else time.monotonic() + timeout
        while True:
            code = self.exit_code()
            if code is not None:
                return code
            if deadline is not None and time.monotonic() >= deadline:
                raise TimeoutError(f"GPU job in {self.run_dir} did not finish in time")
            time.sleep(self.poll_interval)

    def query(self):
        code = self.exit_code()
        return {"state": "exited" if code is not None else
                ("running" if self.alive() else "unknown"),
                "exit_code": code, "pid": self._pid(),
                "run_dir": str(self.run_dir)}

    def cancel(self, *, grace_seconds=30):
        pid = self._pid()
        if pid is None:
            return False
        for signame in ("INT", "TERM", "KILL"):
            remote.run_ssh(self.spec, f"kill -{signame} -- -{pid} 2>/dev/null || "
                                       f"kill -{signame} {pid} 2>/dev/null || true",
                           timeout=self.command_timeout, connect_timeout=self.ssh_timeout)
            deadline = time.monotonic() + grace_seconds
            while time.monotonic() < deadline:
                if self.exit_code() is not None:
                    return True
                time.sleep(min(2.0, self.poll_interval))
        return self.exit_code() is not None

    def statistics(rows, *, threshold=None, worst=8):
    result = {"count": 0, "signed_sum": 0.0, "abs_sum": 0.0, "max_abs_delta": 0.0,
              "min_log_ratio": None, "max_log_ratio": None, "bad_count": 0,
              "histogram": [0] * (len(EDGES) + 1), "worst": []}
    for row_index, (current, behavior, mask) in enumerate(rows):
        if len(current) != len(behavior) or len(mask) != len(current) or not len(current):
            raise ValueError("Probability/mask length mismatch")
        if any(flag not in (0, 1) for flag in mask):
            raise ValueError("Loss mask must be binary")
        for position, (p, q, flag) in enumerate(zip(current, behavior, mask, strict=True)):
            if not finite(p) or not finite(q) or p > 0 or q > 0:
                raise ValueError("Log probabilities must be finite and nonpositive")
            if not flag:
                continue
            delta = float(p) - float(q)
            absolute = abs(delta)
            if not math.isfinite(delta):
                raise ValueError("Nonfinite log-probability difference")
            result["count"] += 1
            result["signed_sum"] += delta
            result["abs_sum"] += absolute
            result["max_abs_delta"] = max(result["max_abs_delta"], absolute)
            result["min_log_ratio"] = delta if result["min_log_ratio"] is None else min(result["min_log_ratio"], delta)
            result["max_log_ratio"] = delta if result["max_log_ratio"] is None else max(result["max_log_ratio"], delta)
            result["bad_count"] += int(threshold is not None and absolute > threshold)
            result["histogram"][bisect.bisect_left(EDGES, absolute)] += 1
            if worst:
                result["worst"].append({"row": row_index, "position": position, "delta": delta})
                result["worst"] = sorted(result["worst"], key=lambda x: abs(x["delta"]), reverse=True)[:worst]
    if not result["count"]:
        raise ValueError("No valid response tokens")
    if not math.isfinite(result["abs_sum"]) or not math.isfinite(result["signed_sum"]):
        raise ValueError("Probability statistics overflow")
    return result

    def merge(parts):
    if not parts:
        raise ValueError("No contributing ranks")
    count = sum(p["count"] for p in parts)
    bins = [sum(p["histogram"][i] for p in parts) for i in range(len(EDGES) + 1)]
    maximum = max(p["max_abs_delta"] for p in parts)
    def percentile(fraction):
        target, seen = math.ceil(count * fraction), 0
        for index, n in enumerate(bins):
            seen += n
            if seen >= target:
                return min(EDGES[index], maximum) if index < len(EDGES) else maximum
        raise ValueError("Corrupt histogram")
    return {"token_count": count, "signed_mean_delta": math.fsum(p["signed_sum"] for p in parts) / count,
            "mean_abs_delta": math.fsum(p["abs_sum"] for p in parts) / count,
            "max_abs_delta": maximum, "p95_abs_delta_upper": percentile(.95), "p99_abs_delta_upper": percentile(.99),
            "min_log_ratio": min(p["min_log_ratio"] for p in parts),
            "max_log_ratio": max(p["max_log_ratio"] for p in parts),
            "bad_fraction": sum(p["bad_count"] for p in parts) / count,
            "percentile_semantics": "logarithmic-histogram upper bound; not exact quantiles",
